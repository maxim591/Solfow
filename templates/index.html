import requests, time

API = "https://api.solfow.com"
KEY = "sk_live_..."

# 1. Create payment
order = requests.post(
    f"{API}/payment",
    headers={"x-api-key": KEY},
    json={
        "address": "YourWalletAddress...",
        "amount": 0.42,
        "redirect_url": "https://yourshop.com/success",
    },
    timeout=10,
).json()

pid = order["url"].split("/")[-1]

# 2. Wait until paid
while True:
    status = requests.get(
        f"{API}/payment/{pid}/status",
        timeout=10,
    ).json()

    if status["paid"]:
        break

    time.sleep(2)

print(f"Paid: {pid}")
